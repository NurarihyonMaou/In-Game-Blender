[CommandListCalcScroll]
local $new_pos

if $move

    if cursor_x >= 0.807 && cursor_x <= 0.807 + 0.005
        if cursor_y >= $scroll_y && cursor_y <= ($scroll_y + 0.1)
            $dragging = 1
            $grab_offset = cursor_y - $scroll_y
        endif
    endif

    if $dragging
    
        $new_pos = cursor_y - $grab_offset
    
        if $new_pos < 0.069
            $new_pos = 0.069
        endif
    
        if $new_pos > (0.92 - 0.1)
            $new_pos = 0.82
        endif
    
        $scroll_y = $new_pos
        $scroll_norm = ($scroll_y - 0.069) / (0.92 - 0.069 - 0.1)
    endif
else
    $dragging = 0
endif
