; Button that brings up or hides the menu
[KeyMenu]
condition = $active == 1
key = no_modifiers /
type = cycle
$menu = 0,1

; Resets menu position
; More specifically, it resets the mod menu widget
; position, made it in case you dragged window somewhere
; where you shouldn't and now can't get it back
; I tried to drag it into some inaccessible place myself
; but couldn't do it, but you can never trust anyone
; not to break it, hence the reset function exists
[KeyMenu.ResetPosition]
condition = $menu == 1
key = no_ctrl shift VK_RIGHT
run = CommandListResetMenuPos

; Moves menu up
[KeyMenu.Menu.UP]
condition = $menu == 1
key = no_alt no_shift ctrl VK_UP
type = hold
run = CommandListMoveMenu.UP

; Moves menu down
[KeyMenu.Menu.DOWN]
condition = $menu == 1
key = no_alt no_shift ctrl VK_DOWN
type = hold
run = CommandListMoveMenu.DOWN

; Moves menu left
[KeyMenu.Menu.LEFT]
condition = $menu == 1
key = no_alt no_shift ctrl VK_LEFT
type = hold
run = CommandListMoveMenu.LEFT

; Moves menu right
[KeyMenu.Menu.RIGHT]
condition = $menu == 1
key = no_alt no_shift ctrl VK_RIGHT
type = hold
run = CommandListMoveMenu.RIGHT

; Navigates forward through the menu (toggles/sliders)
[KeyMenu.Navigate.FORWARD]
condition = $menu == 1 && !$navigating 
key = no_ctrl no_shift no_alt VK_DOWN
type = hold
run = CommandListNavigateMenu.FORWARD

; Navigates backwards through the menu (toggles/sliders)
[KeyMenu.Navigate.BACKWARDS]
condition = $menu == 1 && !$navigating 
key = no_ctrl no_shift no_alt VK_UP
type = hold
run = CommandListNavigateMenu.BACKWARDS

; Emulates click on a selected toggle slot
; Or moves the cursor of selected slider
[KeyMenu.Navigate.PRESS.LEFT]
condition = $menu == 1
key = no_ctrl no_shift no_alt VK_LEFT
type = hold
run = CommandListNavigateMenu.Press.LEFT

[KeyMenu.Navigate.PRESS.RIGHT]
condition = $menu == 1
key = no_ctrl no_shift no_alt VK_RIGHT
type = hold
run = CommandListNavigateMenu.Press.RIGHT

; CommandList -----------------------

[CommandListDimensions]
if window_width != 0 && window_height != 0
  $ww = window_width
  $wh = window_height
elif rt_width != 0 && rt_height != 0
  $ww = rt_width
  $wh = rt_height
else
  $ww = res_width
  $wh = res_height
endif

if cursor_x > 0 && cursor_y > 0 && cursor_x < 1 && cursor_y < 1
    $cpx = cursor_x
    $cpy = cursor_y
elif cursor_window_x > 0 && cursor_window_y > 0 && cursor_window_x < 1 && cursor_window_y < 1
    $cpx = cursor_window_x / $ww
    $cpy = cursor_window_y / $wh
else
    $cpx = cursor_screen_x / $ww
    $cpy = cursor_screen_y / $wh
endif


[CommandListResetMenuPos]
$mx = 50 / $ww
$my = 0.5 - 360 / $wh
$AnimSpeed = 0.2
$TargetAnimSpeed = 0.2

[CommandListMoveMenu]
$mms = $mms + (0.00025 * ($mms < 0.01))
if $menu_d == 1
  $my = $my - $mms
elif $menu_d == 2
  $my = $my + $mms
elif $menu_d == 3
  $mx = $mx - $mms
elif $menu_d == 4
  $mx = $mx + $mms
endif

if $mx < 0
  $mx = 0
elif $mx + 360 / $ww > 1
  $mx = 1 - 360 / $ww
endif

if $my < 0
  $my = 0
elif $my + 560 / $wh > 1
  $my = 1 - 560 / $wh
endif

[CommandListMoveMenu.UP]
$menu_d = 1
post $menu_d = 0
post $mms = 0

[CommandListMoveMenu.DOWN]
$menu_d = 2
post $menu_d = 0
post $mms = 0

[CommandListMoveMenu.LEFT]
$menu_d = 3
post $menu_d = 0
post $mms = 0

[CommandListMoveMenu.RIGHT]
$menu_d = 4
post $menu_d = 0
post $mms = 0

[CommandListNavigateMenu]
$navSpeed = $navSpeed + 0.085
if $navSpeed > 2
  $targetSelectedSlot = ($targetSelectedSlot + ($navDirection * $dt * 6))
  $selectedSlot = $targetSelectedSlot // 1
else
  $targetSelectedSlot = $selectedSlot
endif
if $navDirection == -1
  if $selectedSlot == 0
    $selectedSlot = 9
    $targetSelectedSlot = 10
  endif
elif $navDirection == 1
  if $selectedSlot == 10
    $selectedSlot = 1
    $targetSelectedSlot = 1
  endif
endif

[CommandListNavigateMenu.MoveSliderCursor]
$navSpeed = $navSpeed + (0.02 * $dt * ($navSpeed < 0.02))

if $selectedSlot == 1
  $AnimSpeed = $AnimSpeed + $navSpeed * $navDirection
  if $AnimSpeed < 0
    $AnimSpeed = 0
  elif $AnimSpeed > 1
    $AnimSpeed = 1
  endif
  $TargetAnimSpeed = $AnimSpeed
endif

[CommandListNavigateMenu.FORWARD]
$navDirection = 1
$navigating = 1
$selectedSlot = $selectedSlot + 1
run = CommandListNavigateMenu
post $navDirection = 0
post $navigating = 0
post $navSpeed = 0
post $selectedSlot = $selectedSlot // 1
post $targetSelectedSlot = $selectedSlot

[CommandListNavigateMenu.BACKWARDS]
$navDirection = -1
$navigating = 1
$selectedSlot = $selectedSlot - 1
run = CommandListNavigateMenu
post $navDirection = 0
post $navigating = 0
post $navSpeed = 0
post $selectedSlot = $selectedSlot // 1
post $targetSelectedSlot = $selectedSlot

[CommandListNavigateMenu.Press.LEFT]
if $page == 1
  if $selectedSlot > 0 && $selectedSlot < 7
    $navigating = 2
    $navDirection = -1
    post $navigating = 0
    post $navDirection = 0
    post $navSpeed = 0
    run = CommandListNavigateMenu.MoveSliderCursor
  else
    run = CommandListClickedSlot
  endif
  
else
  run = CommandListClickedSlot
endif

[CommandListNavigateMenu.Press.RIGHT]
if $page == 1
  if $selectedSlot > 0 && $selectedSlot < 7
    $navigating = 2
    $navDirection = 1
    post $navigating = 0
    post $navDirection = 0
    post $navSpeed = 0
    run = CommandListNavigateMenu.MoveSliderCursor
  else
    run = CommandListClickedSlot
  endif
else
  run = CommandListClickedSlot
endif

[CommandListBackground]
local $cox
local $coy
ps-t100 = ResourceMenuBG
y87 = 560 / $wh
x87 = 360 / $ww
if $hovering == 0
    if $drag == 0
        if $cpx > $mx + 0.0125 && $cpx < ($mx+x87-0.0035) && $cpy > $my && $cpy < ($my+y87-0.0035)
            if $hold
                $cox = $cpx - $mx
                $coy = $cpy - $my
                $drag = 1
            endif
        endif
    else
        if $hold
            $mx = $cpx - $cox
            $my = $cpy - $coy
        else
            $drag = 0
        endif
    endif
endif
z87 = $mx
w87 = $my
run = CustomShaderElement
run = CommandListDrawToggles

[CommandListSlot]

run = CommandListSlotItemImage
if $cpx > z87 && $cpx < (z87+x87-0.0035) && $cpy > w87 && $cpy < (w87+y87-0.0035)
    ps-t100 = ResourceItemSlotHover.1
    $hovering = 1
    $hoveredSlot = $slot
    $selectedSlot = 0
else
    if $slot == $selectedSlot // 1
      $hoveredSlot = $slot
      ps-t100 = ResourceItemSlotHover.1
    else
      ps-t100 = ResourceItemSlot
    endif
endif
run = CustomShaderElement

    ps-t100 = ResourceItemSlotHover.2
    run = CustomShaderElement
    $clickedSlot = 0
endif
